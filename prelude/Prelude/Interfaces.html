258 | ||| The `Foldable` interface describes how you can iterate over the elements in
259 | ||| a parameterised type and combine the elements together, using a provided
260 | ||| function, into a single result.

262 | ||| A minimal definition includes `foldr`

265 |   constructor MkFoldable
266 |   ||| Successively combine the elements in a parameterised type using the
267 |   ||| provided function, starting with the element that is in the final position
268 |   ||| i.e. the right-most position.
269 |   ||| @ func  The function used to 'fold' an element into the accumulated result
270 |   ||| @ init  The starting value the results are being combined into
271 |   ||| @ input The parameterised type
272 |   foldr : (func : elem -> acc -> acc) -> (init : acc) -> (input : t elem) -> acc
273 |
274 |   ||| The same as `foldr` but begins the folding from the element at the initial
275 |   ||| position in the data structure i.e. the left-most position.
276 |   ||| @ func  The function used to 'fold' an element into the accumulated result
277 |   ||| @ init  The starting value the results are being combined into
278 |   ||| @ input The parameterised type
279 |   foldl : (func : acc -> elem -> acc) -> (init : acc) -> (input : t elem) -> acc
280 |   foldl f z t = foldr (flip (.) . flip f) id t z
281 |
282 |   ||| Test whether the structure is empty.
283 |   ||| @ acc The accumulator value which is specified to be lazy
284 |   null : t elem -> Bool
285 |   null xs = foldr {acc = Lazy Bool} (\ _,_ => False) True xs
286 |
287 |   ||| Similar to `foldl`, but uses a function wrapping its result in a `Monad`.
288 |   ||| Consequently, the final value is wrapped in the same `Monad`.
289 |   foldlM : Monad m => (funcM : acc -> elem -> m acc) -> (init : acc) -> (input : t elem) -> m acc

294 |   toList = foldr (::) []
295 |
296 |   ||| Maps each element to a value and combine them.
297 |   ||| For performance reasons, this should wherever
