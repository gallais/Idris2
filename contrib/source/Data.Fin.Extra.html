 14 | ||| A Fin's underlying natural number is smaller than the bound
 15 | export
 16 | elemSmallerThanBound : (n : Fin m) -> LT (finToNat n) m
 17 | elemSmallerThanBound FZ = LTESucc LTEZero
 18 | elemSmallerThanBound (FS x) = LTESucc (elemSmallerThanBound x)
 19 |
 20 | ||| Last's underlying natural number is the bound's predecessor
 21 | export
 22 | finToNatLastIsBound : {n : Nat} -> finToNat (Fin.last {n}) = n
 23 | finToNatLastIsBound {n=Z} = Refl

 25 |
 26 | ||| Weaken does not modify the underlying natural number
 27 | export
 28 | finToNatWeakenNeutral : {n : Fin m} -> finToNat (weaken n) = finToNat n
 29 | finToNatWeakenNeutral = finToNatQuotient (weakenNeutral n)
 30 |
 31 | ||| WeakenN does not modify the underlying natural number
 32 | export
 33 | finToNatWeakenNNeutral : (0 m : Nat) -> (k : Fin n) ->
 34 |                          finToNat (weakenN m k) = finToNat k
 35 | finToNatWeakenNNeutral m k = finToNatQuotient (weakenNNeutral m k)
 36 |
 37 | ||| `Shift k` shifts the underlying natural number by `k`.
 38 | export
 39 | finToNatShift : (k : Nat) -> (a : Fin n) -> finToNat (shift k a) = k + finToNat a
 40 | finToNatShift Z     a = Refl
 41 | finToNatShift (S k) a = cong S (finToNatShift k a)

 46 |
 47 | ||| Compute the Fin such that `k + invFin k = n - 1`
 48 | public export
 49 | invFin : {n : Nat} -> Fin n -> Fin n
 50 | invFin FZ = last
 51 | invFin (FS k) = weaken (invFin k)
 52 |
 53 | export
 54 | invFinSpec : {n : _} -> (i : Fin n) -> 1 + finToNat i + finToNat (invFin i) = n
 55 | invFinSpec {n = S k} FZ = cong S finToNatLastIsBound
 56 | invFinSpec (FS k) = let H = invFinSpec k in
 57 |   let h = finToNatWeakenNeutral {n = invFin k} in
 58 |   cong S (rewrite h in H)
 59 |
 60 | ||| The inverse of a weakened element is the successor of its inverse
 61 | export
 62 | invFinWeakenIsFS : {n : Nat} -> (m : Fin n) -> invFin (weaken m) = FS (invFin m)
 63 | invFinWeakenIsFS FZ = Refl
 64 | invFinWeakenIsFS (FS k) = cong weaken (invFinWeakenIsFS k)
 65 |
 66 | export
 67 | invFinLastIsFZ : {n : Nat} -> invFin (last {n}) = FZ
 68 | invFinLastIsFZ {n = Z} = Refl
 69 | invFinLastIsFZ {n = S n} = cong weaken (invFinLastIsFZ {n})
 70 |
 71 | ||| `invFin` is involutive (i.e. applied twice it is the identity)
 72 | export
 73 | invFinInvolutive : {n : Nat} -> (m : Fin n) -> invFin (invFin m) = m
 74 | invFinInvolutive FZ = invFinLastIsFZ
 75 | invFinInvolutive (FS k) = Calc $
 76 |    |~ invFin (invFin (FS k))
 77 |    ~~ invFin (weaken (invFin k)) ...( Refl )
 78 |    ~~ FS (invFin (invFin k))     ...( invFinWeakenIsFS (invFin k) )
 79 |    ~~ FS k                       ...( cong FS (invFinInvolutive k) )

136 | ||| A view of Nat as a quotient of some number and a finite remainder.
137 | public export
138 | data FractionView : (n : Nat) -> (d : Nat) -> Type where
139 |     Fraction : (n : Nat) -> (d : Nat) -> {auto ok: GT d Z} ->
140 |                 (q : Nat) -> (r : Fin d) ->
141 |                 q * d + finToNat r = n -> FractionView n d
142 |
143 | ||| Converts Nat to the fractional view with a non-zero divisor.
144 | export
145 | divMod : (n, d : Nat) -> {auto ok: GT d Z} -> FractionView n d
146 | divMod Z (S d) = Fraction Z (S d) Z FZ Refl
147 | divMod {ok=_} (S n) (S d) =
148 |     let Fraction {ok=ok} n (S d) q r eq = divMod n (S d) in
149 |     case strengthen' r of
150 |         Left eq' => Fraction {ok=ok} (S n) (S d) (S q) FZ $
151 |             rewrite sym eq in
152 |                 rewrite trans (cong finToNat eq') finToNatLastIsBound in
153 |                     cong S $ trans
154 |                         (plusZeroRightNeutral (d + q * S d))
155 |                         (plusCommutative d (q * S d))
156 |         Right (r' ** eq') => Fraction {ok=ok} (S n) (S d) q (FS r') $
157 |             rewrite sym $ plusSuccRightSucc (q * S d) (finToNat r') in
158 |                 cong S $ trans (sym $ cong (plus (q * S d)) eq') eq

335 | ||| Converts `Fin`s that are used as indexes of parts to an index of a sum.
336 | |||
337 | ||| For example, if you have a `Vect` that is a concatenation of two `Vect`s and
338 | ||| you have an index either in the first or the second of the original `Vect`s,
339 | ||| using this function you can get an index in the concatenated one.

343 | indexSum (Right r) = shift m r

345 | ||| Extracts an index of the first or the second part from the index of a sum.
346 | |||
347 | ||| For example, if you have a `Vect` that is a concatenation of the `Vect`s and
348 | ||| you have an index of this `Vect`, you have get an index of either left or right
349 | ||| original `Vect` using this function.

355 |
356 | ||| Calculates the index of a square matrix of size `a * b` by indices of each side.
357 | public export
358 | indexProd : {n : Nat} -> Fin m -> Fin n -> Fin (m * n)
