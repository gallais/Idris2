 2 | ||| by Liam O'Connor
 3 | |||
 4 | ||| The main difference is that we use `Colist1` for the type of
 5 | ||| generators rather than `Stream`. This allows us to avoid generating
 6 | ||| many duplicates when dealing with finite types which are common in
 7 | ||| programming (Bool) but even more so in dependently typed programming
 8 | ||| (Vect 0, Fin (S n), etc.).
 9 |
10 | module Search.Generator
11 |
12 | import Data.Colist
13 | import Data.Colist1
14 | import Data.Fin
15 | import Data.Stream
16 | import Data.Vect
17 |
18 | ------------------------------------------------------------------------
19 | -- Interface
20 |
21 | ||| A generator for a given type is a non-empty colist of values of that
22 | ||| type.
23 | public export
24 | interface Generator a where
25 |   generate : Colist1 a
26 |
27 | ------------------------------------------------------------------------
28 | -- Implementations
29 |
30 | -- Finite types
31 |
32 | ||| ALL of the natural numbers
33 | public export
34 | Generator Nat where
35 |   generate = fromStream nats
36 |
37 | ||| ALL of the booleans
38 | public export
39 | Generator Bool where
40 |   generate = True ::: [False]
41 |
42 | ||| ALL of the Fins
43 | public export
44 | {n : Nat} -> Generator (Fin (S n)) where
45 |   generate = fromList1 (allFins n)
46 |
47 | -- Polymorphic generators
48 |
49 | ||| We typically want to generate a generator for a unit-terminated right-nested
50 | ||| product so we have this special case that keeps the generator minimal.
51 | public export
52 | Generator a => Generator (a, ()) where
53 |   generate = map (,()) generate
54 |
55 | ||| Put two generators together by exploring the plane they define.
56 | ||| This uses Cantor's zig zag traversal
57 | public export
58 | (Generator a, Generator b) => Generator (a, b) where
59 |   generate = plane generate (\ _ => generate)
60 |
61 | ||| Put two generators together by exploring the plane they define.
62 | ||| This uses Cantor's zig zag traversal
63 | public export
64 | {0 b : a -> Type} -> (Generator a, (x : a) -> Generator (b x)) =>
65 |   Generator (x : a ** b x) where
66 |   generate = plane generate (\ x => generate)
67 |
68 | ||| Build entire vectors of values
69 | public export
70 | {n : Nat} -> Generator a => Generator (Vect n a) where
71 |   generate = case n of
72 |     Z => [] ::: []
73 |     S n => map (uncurry (::)) generate
74 |
75 | ||| Generate arbitrary lists of values
76 | ||| Departing from the paper, to avoid having infinitely many copies of
77 | ||| the empty list, we handle the case `n = 0` separately.
78 | public export
79 | Generator a => Generator (List a) where
80 |   generate = [] ::: forget (planeWith listy generate vectors) where
81 |     listy : (n : Nat) -> Vect (S n) a -> List a
82 |     listy _ = toList
83 |
84 |     vectors : (n : Nat) -> Colist1 (Vect (S n) a)
85 |     vectors n = generate
