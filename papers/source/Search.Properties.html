 26 | ||| Take the product of a list of types
 27 | public export
 28 | Product : List Type -> Type
 29 | Product = foldr Pair ()
 30 |
 31 | ||| A property amenable to testing
 32 | ||| @cs is the list of generators we need (inferrable)
 33 | ||| @a  is the type we hope is inhabited
 34 | ||| NB: the longer the list of generators, the bigger the search space!
 35 | public export
 36 | record Prop (cs : List Type) (a : Type) where
 37 |   constructor MkProp
 38 |   ||| The function trying to find an `a` provided generators for `cs`.
 39 |   ||| Made total by consuming some fuel along the way.
 40 |   runProp : Colist1 (Product cs) -> Fuel -> HDec a
 41 |
 42 | ------------------------------------------------------------------------
 43 | -- Prop-like structure
 44 |
 45 | ||| A type constructor satisfying the AProp interface is morally a Prop
 46 | ||| It may not make use of all of the powers granted by Prop, hence the
 47 | ||| associated `Constraints` list of types.
 48 | public export
 49 | interface AProp (0 t : Type -> Type) where
 50 |   0 Constraints : List Type
 51 |   toProp : t a -> Prop Constraints a
 52 |
 53 | ||| Props are trivially AProp
 54 | public export
 55 | AProp (Prop cs) where
 56 |   Constraints = cs
 57 |   toProp = id
 58 |
 59 | ||| Half deciders are AProps that do not need any constraints to be satisfied
 60 | public export
 61 | AProp HDec where
 62 |   Constraints = []
 63 |   toProp = MkProp . const . const
 64 |
 65 | ||| Deciders are AProps that do not need any constraints to be satisfied
 66 | public export
 67 | AProp Dec where
 68 |   Constraints = []
 69 |   toProp = MkProp . (const . const . toHDec)
 70 |
 71 | ||| We can run an AProp to try to generate a value of type a
 72 | public export
 73 | check : (gen : Generator (Product cs)) => (f : Fuel) -> (p : Prop cs a) ->
 74 |         {auto pr : So (isTrue (runProp p (generate @{gen}) f))} -> a
 75 | check @{gen} f p @{pr} = evidence (runProp p (generate @{gen}) f) pr
 76 |
 77 | ||| Provided that we know how to generate candidates of type `a`, we can look
 78 | ||| for a witness satisfying a given predicate over `a`.
 79 | public export
 80 | exists : {0 p : a -> Type} -> (aPropt : AProp t) =>
 81 |          ((x : a) -> t (p x)) ->
 82 |          Prop (a :: Constraints @{aPropt}) (DPair a p)
 83 | exists test = MkProp $ \ acs, fuel =>
 84 |   let candidates : LazyList a = take fuel (map fst acs) in
 85 |   let cs = map snd acs in
 86 |   let find = any candidates (\ x => runProp (toProp (test x)) cs fuel) in
 87 |   map toDPair find

102 |   formula : Type
103 |   formula = DPair Nat $ \ m =>
104 |             DPair Nat $ \ n =>
105 |             DPair Nat $ \ p =>
106 |             (0 `LT` m, 0 `LT` n, 0 `LT` p
107 |             , (m * m + n * n) === (p * p))
108 |
109 |   search : Prop ? Pythagoras.formula

111 |            exists $ \ n =>
112 |            exists $ \ p =>
113 |            (isLT 0 m && isLT 0 n && isLT 0 p
114 |            && decEq (m * m + n * n) (p * p))
