 30 |   shiftL : a -> Index -> a

 33 |   shiftR : a -> Index -> a
 34 |

 36 |   bit : (i : Index) -> a
 37 |
 38 |   ||| The value with all bits unset.
 39 |   zeroBits : a
 40 |
 41 |   ||| Returns the bitwise complement of a value.
 42 |   complement : a -> a
 43 |   complement = xor oneBits
 44 |
 45 |   ||| The value with all bits set..
 46 |   oneBits : a
 47 |   oneBits = complement zeroBits
 48 |
 49 |   ||| `complementBit x i` is the same as `xor x (bit i)`.
 50 |   complementBit : (x : a) -> (i : Index) -> a
 51 |   complementBit x i = x `xor` bit i
 52 |
 53 |   ||| `clearBit x i` is the same as `x .&. complement (bit i)`
 54 |   clearBit : (x : a) -> (i : Index) -> a
 55 |   clearBit x i = x `xor` (bit i .&. x)
 56 |
 57 |   ||| Tests, whether the i-th bit is set in the given value.
 58 |   testBit : a -> Index -> Bool
 59 |
 60 |   ||| Sets the i-th bit of a value.
 61 |   setBit : a -> (i : Index) -> a
 62 |   setBit x i = x .|. bit i
