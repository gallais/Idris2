 82 | public export
 83 | (DecEq a, DecEq b) => DecEq (a, b) where
 84 |   decEq (a, b) (a', b') with (decEq a a')
 85 |     decEq (a, b) (a', b') | (No contra) =
 86 |       No $ contra . fst . pairInjective
 87 |     decEq (a, b) (a, b') | (Yes Refl) with (decEq b b')
 88 |       decEq (a, b) (a, b) | (Yes Refl) | (Yes Refl) = Yes Refl
 89 |       decEq (a, b) (a, b') | (Yes Refl) | (No contra) =
 90 |         No $ contra . snd . pairInjective

125 | -- For the primitives, we have to cheat because we don't have access to their
126 | -- internal implementations. We use believe_me for the inequality proofs

132 | ||| An unsafe decidable equality implementation based on boolean equality.
133 | ||| Useful for builtin types.
134 | public export
135 | [FromEq] Eq a => DecEq a where
136 |     decEq x y = case x == y of -- Blocks if x or y not concrete
137 |                      True => Yes primitiveEq
138 |                      False => No primitiveNotEq
139 |        where primitiveEq : forall x, y . x = y
140 |              primitiveEq = believe_me (Refl {x})
141 |              primitiveNotEq : forall x, y . Not (x = y)
142 |              primitiveNotEq prf = believe_me {b = Void} ()
