181 | ||| Collect the given signal.
182 | |||
183 | ||| This replaces the existing handling of the given signal
184 | ||| and instead results in Idris collecting occurrences of
185 | ||| the signal until you call `handleNextCollectedSignal`.
186 | |||
187 | ||| First, call `collectSignal` for any number of signals.
188 | ||| Then, call `handleNextCollectedSignal` in each main loop
189 | ||| of your program to retrieve (and mark as handled) the next
190 | ||| signal that was collected, if any.
191 | |||
192 | ||| Signals are not queued, so the return order of signals is
193 | ||| not specified and signals may be deduplicated.
194 | export
195 | collectSignal : HasIO io => Signal -> io (Either SignalError ())
196 | collectSignal sig = do
197 |   res <- primIO $ prim__collectSignal (signalCode sig)
198 |   case isError res of
199 |        False => pure $ Right ()
200 |        True  => Left <$> getError
201 |
202 | ||| Get next collected signal under the pretense of handling it.
203 | |||
204 | ||| Calling this "marks" the signal as handled so the next time
205 | ||| this function is called you will retrieve the next unhandled
206 | ||| signal instead of the same signal again.
207 | |||
208 | ||| You get back Nothing if there are no pending signals.
209 | export
210 | handleNextCollectedSignal : HasIO io => io (Maybe Signal)
211 | handleNextCollectedSignal =
212 |   toSignal <$> primIO prim__handleNextCollectedSignal
213 |
214 | ||| Get many collected signals and mark them as handled.
215 | |||
216 | ||| Use `forever` as your fuel if you don't want or need to
217 | ||| retain totality. Alternatively, pick a max number to
218 | ||| retrieve and use `limit/1` as your fuel.
219 | export
220 | handleManyCollectedSignals : HasIO io => Fuel -> io (List Signal)
