11 | ||| A proof that some element is found in a list.
12 | public export
13 | data Elem : a -> List a -> Type where
14 |      ||| A proof that the element is at the head of the list
15 |      Here : Elem x (x :: xs)
16 |      ||| A proof that the element is in the tail of the list
17 |      There : Elem x xs -> Elem x (y :: xs)
18 |
19 | export
20 | Uninhabited (Here = There e) where
21 |   uninhabited Refl impossible
22 |
23 | export
24 | Uninhabited (There e = Here) where
25 |   uninhabited Refl impossible
26 |
27 | export
28 | Injective (There {x} {y} {xs}) where
29 |   injective Refl = Refl
30 |
31 | export
32 | DecEq (Elem x xs) where
33 |   decEq Here Here = Yes Refl
34 |   decEq Here (There later) = No absurd
35 |   decEq (There later) Here = No absurd
36 |   decEq (There this) (There that) with (decEq this that)
37 |     decEq (There this) (There this) | Yes Refl  = Yes Refl
38 |     decEq (There this) (There that) | No contra = No (contra . injective)
39 |
40 | export
41 | Uninhabited (Elem {a} x []) where
42 |   uninhabited Here impossible
43 |   uninhabited (There p) impossible
44 |
45 | export
46 | Uninhabited (x = z) => Uninhabited (Elem z xs) => Uninhabited (Elem z $ x::xs) where
47 |   uninhabited Here @{xz} = uninhabited Refl @{xz}
48 |   uninhabited (There y) = uninhabited y
49 |
50 | ||| An item not in the head and not in the tail is not in the list at all.
51 | export
52 | neitherHereNorThere : Not (x = y) -> Not (Elem x xs) -> Not (Elem x (y :: xs))
53 | neitherHereNorThere xny _     Here        = xny Refl
54 | neitherHereNorThere _   xnxs  (There xxs) = xnxs xxs
55 |
56 | ||| Check whether the given element is a member of the given list.
57 | public export
58 | isElem : DecEq a => (x : a) -> (xs : List a) -> Dec (Elem x xs)
59 | isElem x [] = No absurd
60 | isElem x (y :: xs) with (decEq x y)
61 |   isElem x (x :: xs) | Yes Refl = Yes Here
62 |   isElem x (y :: xs) | No xny with (isElem x xs)
63 |     isElem x (y :: xs) | No xny | Yes xxs = Yes (There xxs)
64 |     isElem x (y :: xs) | No xny | No xnxs = No (neitherHereNorThere xny xnxs)
65 |
66 | ||| Remove the element at the given position.
67 | public export
68 | dropElem : (xs : List a) -> (p : Elem x xs) -> List a
69 | dropElem (_ :: ys)  Here     = ys
70 | dropElem (x :: ys) (There p) = x :: dropElem ys p
71 |
72 | ||| Erase the indices, returning the numeric position of the element
73 | public export
74 | elemToNat : Elem x xs -> Nat
75 | elemToNat  Here     = Z
76 | elemToNat (There p) = S (elemToNat p)
77 |
78 | ||| Find the element with a proof at a given position, if it is valid
79 | public export
80 | indexElem : Nat -> (xs : List a) -> Maybe (x ** Elem x xs)
81 | indexElem  _    []        = Nothing
82 | indexElem  Z    (y :: _)  = Just (y ** Here)
83 | indexElem (S n) (_ :: ys) = map (\(x ** p) => (x ** There p)) (indexElem n ys)
84 |
85 | ||| Lift the membership proof to a mapped list
86 | export
87 | elemMap : (0 f : a -> b) -> Elem x xs -> Elem (f x) (map f xs)
88 | elemMap f  Here      = Here
89 | elemMap f (There el) = There $ elemMap f el
