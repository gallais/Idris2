36 | neitherHereNorThere : Not (x = y) -> Not (Elem x xs) -> Not (Elem x (y :: xs))
37 | neitherHereNorThere xneqy xninxs  Here         = xneqy Refl
38 | neitherHereNorThere xneqy xninxs (There xinxs) = xninxs xinxs

40 | ||| A decision procedure for Elem
41 | public export
42 | isElem : DecEq a => (x : a) -> (xs : Vect n a) -> Dec (Elem x xs)
43 | isElem x [] = No uninhabited
44 | isElem x (y::xs) with (decEq x y)
45 |   isElem x (x::xs) | (Yes Refl) = Yes Here
46 |   isElem x (y::xs) | (No xneqy) with (isElem x xs)
47 |     isElem x (y::xs) | (No xneqy) | (Yes xinxs) = Yes (There xinxs)
48 |     isElem x (y::xs) | (No xneqy) | (No xninxs) = No (neitherHereNorThere xneqy xninxs)
